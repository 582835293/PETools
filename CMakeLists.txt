cmake_minimum_required(VERSION 3.20)
project(PETools VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "PETools currently requires a Windows x86 build (-A Win32).")
endif()
add_library(petools STATIC PETools.c)
target_include_directories(petools PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(petools PUBLIC UNICODE _UNICODE)
if(MSVC)
  target_compile_options(petools PUBLIC /utf-8)
endif()
option(PETOOLS_BUILD_TESTS "Build PE32 fixture and C++ linkage tests" ${PROJECT_IS_TOP_LEVEL})
if(PETOOLS_BUILD_TESTS)
  enable_testing()
  add_executable(petools_tests tests/petools_tests.c)
  target_link_libraries(petools_tests PRIVATE petools)
  add_test(NAME petools_fixture COMMAND petools_tests)
  add_executable(petools_cpp_link tests/cpp_link_smoke.cpp)
  target_link_libraries(petools_cpp_link PRIVATE petools)
  add_test(NAME petools_cpp_link COMMAND petools_cpp_link)
endif()
